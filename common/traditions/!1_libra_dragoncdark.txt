tr_libra_dragondark_adopt = {
	unlocks_agenda = agenda_libra_dragondark
	possible = {
		if = {
			limit = { has_tradition = tr_libra_dragonlight_adopt }
			num_ascension_perks > 3
			custom_tooltip = {
				fail_text = "tr_libra_dragonborn_fail_2"
				has_tradition = tr_libra_dragonlight_finish
				libra_has_finished_organic_ascension_path = yes
			}
		}
		else = {
			num_ascension_perks > 1
			custom_tooltip = {
				fail_text = "tr_libra_dragonborn_fail_1"
				has_technology = tech_libra_dragon_breath_1
				has_tradition = tr_libra_dragonchant_finish
			}
		}
	}
	on_enabled = {
		hidden_effect = {
			if = {
				limit = { has_tradition = tr_libra_dragonlight_adopt }
				country_event = { id = libra_origin.0728 }
			}
			else = {
				country_event = { id = libra_origin.0725 }
			}
		}
	}
	modifier = {
		country_starbase_capacity_mult = 0.2
	}
}
