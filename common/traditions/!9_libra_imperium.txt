tr_libra_imperium_adopt = {
	unlocks_agenda = agenda_libra_imperium_a
	on_enabled = {
		hidden_effect = {
			if = {
				limit = { exists = ruler }
				ruler = {
					switch = {
						trigger = leader_class
						official = { root = { set_country_flag = libra_imperium_ruler_official } }
						commander = { root = { set_country_flag = libra_imperium_ruler_commander } }
						scientist = { root = { set_country_flag = libra_imperium_ruler_scientist } }
					}
				}
			}
		}
	}
	modifier = {
		edicts_cost_mult = -0.15
	}
	on_disabled = {
		remove_country_flag = libra_imperium_ruler_official
		remove_country_flag = libra_imperium_ruler_commander
		remove_country_flag = libra_imperium_ruler_scientist
	}
	possible = {
		has_technology = tech_planetary_unification
	}
}

tr_libra_imperium_finish = {
