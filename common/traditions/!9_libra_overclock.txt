tr_libra_overclock_adopt = {
	unlocks_agenda = agenda_libra_overclock
	possible = {
		has_technology = tech_capital_productivity_1
	}
	modifier = {
		country_alloys_produces_mult = 0.05
		country_consumer_goods_produces_mult = 0.05
	}
	on_disabled = {
		libra_clear_overclock_modifiers = yes
	}
}

tr_libra_overclock_finish = {
	on_enabled = {
		hidden_effect = {
			if = {
				limit = { is_ai = no }
				set_policy = { policy = libra_overclock_tradition option = policy_libra_overclock_tradition_b cooldown = no }
			}
		}
	}
	modifier = {
		ascension_perks_add = 1
	}
	custom_tooltip_with_modifiers = tr_libra_overclock_finish_desc
}
