	unlocks_agenda = agenda_libra_occlusion
	on_enabled = {
		hidden_effect = {
			every_owned_leader = {
				limit = {
					or = {
						leader_class = commander
						leader_class = official
					}

					trait = random_common
					show_message = no
				}
				change_variable = {
					which = libra_extra_starting_traits_count
					value = 1
				}
			}
			every_pool_leader = {
				limit = {
					or = {
						leader_class = commander
						leader_class = official
					}
				}
				add_trait = {
					trait = random_common
					show_message = no

					value = 1
				}
			}

				value = 1

	}
	possible = {
		has_technology = tech_encryption_2
	}
}
