tr_libra_crimson_ancestor_adopt = {
	unlocks_agenda = agenda_libra_crimson_ancestor
	possible = {
		#has_technology = tech_libra_vampire_beam_1
		custom_tooltip = {
			fail_text = "tr_libra_crimson_ancestor_fail_1"
			has_tradition = tr_libra_crimson_vampire_finish
			or = {
				has_country_flag = libra_crimson_vampire_peace
				has_country_flag = libra_crimson_vampire_war
			}
		}
		if = {
			limit = { is_natural_design_empire = yes }
			num_ascension_perks > 2
		}
		else = {
			custom_tooltip = {
				fail_text = "tr_libra_crimson_ancestor_fail_2"
				libra_has_finished_organic_ascension_path = yes
			}
		}
	}
	custom_tooltip = tr_libra_crimson_ancestor_desc
	on_enabled = {
		country_event = { id = libra_origin.0540 days = 1 }
	}
}
