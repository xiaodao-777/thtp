	unlocks_agenda = agenda_libra_victory

		damage_vs_player_crisis_mult = 0.25
	}
	tradition_swap = {
		inherit_name = yes
		inherit_icon = yes
		trigger = {
			or = {
				has_tradition = tr_libra_pendragon_adopt

			}
		}
		modifier = {
			damage_vs_superior_empires_mult = 0.25
			damage_vs_player_crisis_mult = 0.5
		}
	}
	tradition_swap = {
		inherit_name = yes
		inherit_icon = yes
		trigger = {
			has_tradition = tr_libra_pendragon_adopt
			has_tradition = tr_libra_havoc_adopt
		}
		modifier = {
			damage_vs_superior_empires_mult = 0.25
			damage_vs_player_crisis_mult = 0.75
		}
	}
	on_disabled = {
		every_owned_leader = {

		}

	possible = {

		}
	}
