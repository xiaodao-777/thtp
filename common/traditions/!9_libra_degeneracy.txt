tr_libra_degeneracy_adopt = {
	unlocks_agenda = agenda_libra_degeneracy_a
	on_enabled = {
		hidden_effect = {
			set_variable = { which = libra_degeneracy_decision_count value = 3 }
		}
	}
	modifier = {
		planet_housing_mult = 0.05
	}
	possible = {
		if = {
			limit = {
				or = {
					has_ethic = ethic_gestalt_consciousness
					has_civic = civic_dystopian_society
					has_country_flag = libra_is_machine_empire
				}
			}
			has_technology = tech_society_2
		}
		else = {
			has_technology = tech_subdermal_stimulation
		}
	}
}
