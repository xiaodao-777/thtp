tr_libra_grandmastery_adopt = {
	unlocks_agenda = agenda_libra_grandmastery_a
	custom_tooltip = tr_libra_grandmastery_adopt_desc
	on_enabled = {
		hidden_effect = {
			every_owned_leader = {
				limit = {
					has_base_skill < 4
					can_have_libra_special_leader_traits = yes
				}
				add_skill_without_trait_selection = 1
			}
		}
	}
	on_disabled = {
		every_owned_leader = {
			remove_trait = leader_trait_libra_grandmastery_commander
			remove_trait = leader_trait_libra_grandmastery_commander_max
			remove_trait = leader_trait_libra_grandmastery_official
			remove_trait = leader_trait_libra_grandmastery_official_max
			remove_trait = leader_trait_libra_grandmastery_scientist
			remove_trait = leader_trait_libra_grandmastery_scientist_max
		}
		remove_modifier = libra_grandmastery_5_modifier
		clear_variable = libra_grandmastery_count
	}
	possible = {
		if = {
			limit = {
				has_ethic = ethic_gestalt_consciousness
			}
			has_technology = tech_embodied_dynamism
		}
		else = {
			has_technology = tech_collective_self
		}
	}
}

tr_libra_grandmastery_finish = {

	}
	custom_tooltip_with_modifiers = tr_libra_grandmastery_finish_desc
}

tr_libra_grandmastery_1 = {
	on_enabled = {
		hidden_effect = {
			ordered_owned_leader = {
				limit = {
					leader_class = commander
					nor = {
						has_trait = leader_trait_libra_grandmastery_commander
						has_trait = leader_trait_libra_grandmastery_commander_max
					}
					can_have_libra_special_leader_traits = yes
				}
				position = 0
				order_by = trigger:leader_years_of_service
				if = {
					limit = { leader_years_of_service > 49 }
					add_trait = {
						trait = leader_trait_libra_grandmastery_commander_max
						show_message = yes
					}
				}
				else = {
					add_trait = {
						trait = leader_trait_libra_grandmastery_commander
						show_message = yes
					}
				}
			}
			change_variable = {
				which = libra_grandmastery_count
				value = 1
			}
		}
	}
	modifier = {
		council_agenda_progress_speed = 0.1
		commander_skill_add = 1
	}
	custom_tooltip_with_modifiers = tr_libra_grandmastery_1_desc

	ai_weight = {
		factor = 3000
	}
}

tr_libra_grandmastery_5 = {
	possible = {
		has_tradition = tr_libra_grandmastery_2
		has_tradition = tr_libra_grandmastery_3
	}
	on_enabled = {
		hidden_effect = {
			if = {
				limit = {
					is_variable_set = libra_grandmastery_count
					check_variable = {
						which = libra_grandmastery_count
						value < 15
					}
				}
				add_modifier = {
					modifier = libra_grandmastery_5_modifier
					multiplier = libra_grandmastery_count
				}
			}
			else_if = {
				limit = { is_variable_set = libra_grandmastery_count }
				add_modifier = {
					modifier = libra_grandmastery_5_modifier
					multiplier = 15
				}
			}
		}
	}
	modifier = {
		ruler_skill_add = 2
	}
	custom_tooltip_with_modifiers = tr_libra_grandmastery_5_desc

	ai_weight = {
		factor = 3000
	}
}
