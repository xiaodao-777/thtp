decision_libra_shield_machine = {
	owned_planets_only = yes
	icon = decision_deal

	potential = {
		exists = owner
		owner = {
			has_ascension_perk = ap_libra_shield_machine
		}
	}

	allow = {
		custom_tooltip = {
			fail_text = "requires_leader_not_libra_shield_machine"
			owner = { is_ai = no }
			nand = {
				exists = leader
				leader = { has_trait = leader_trait_libra_shield_machine }
			}
		}
	}

	resources = {
		category = decisions
		cost = {
			energy = 1000
		}
	}

	effect = {
		if = {
			limit = {
				owner = { any_owned_leader = { has_trait = leader_trait_libra_shield_machine } }
			}
			custom_tooltip = decision_libra_shield_machine_effect_a
		}
		else = {
			custom_tooltip = decision_libra_shield_machine_effect_b
		}
		hidden_effect = {
			planet_event = { id = libra_fairy.4010 }
		}
	}

	ai_weight = {
		base = 0
		modifier = {
			add = 0.01
			owner = {
				not = { has_country_flag = libra_shield_machine_moved }
			}
		}
		modifier = {
			factor = trigger:sapient_pop_amount
			or = {
				libra_has_farming_designation = yes
				libra_has_generator_designation = yes
				libra_has_mining_designation = yes
				libra_has_rural_designation = yes
			}
		}
	}
