libra_chromosome_engineer = {
	category = specialist

	swappable_data = {
		default = {
			condition_string = SPECIALIST_JOB_TRIGGER
			building_icon = building_clone_vats
		}
	}

	tags = { research pop_assembly }

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		complex_specialist_job_check_trigger = yes
	}

	resources = {
		category = planet_libra_chromosome_engineer
		produces = {
			society_research = 2
			trade = 6
		}
		upkeep = {
			consumer_goods = 2
		}
	}

	planet_modifier = {
		pop_environment_tolerance = 0.025
		fake_pop_growth_mod = 1
	}
	triggered_planet_pop_group_modifier_for_species = {
		potential = {
			nor = {
				has_trait = trait_psionic
				has_trait = trait_latent_psionic
				has_trait = trait_cybernetic
				has_trait = trait_limited_cybernetic
			}
		}
		bonus_pop_growth = 1
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_libra_genetic_flexible_a
				has_country_flag = genetic_ascension_stage_1_mutation
				not = { has_country_flag = genetic_ascension_stage_1_purity }
			}
		}
		bonus_pop_growth_mult = 0.025
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_libra_genetic_flexible_a
				has_country_flag = genetic_ascension_stage_1_purity
			}
		}
		fake_pop_growth_mod = 0.5
	}
	triggered_planet_pop_group_modifier_for_all = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_libra_genetic_flexible_a
				has_country_flag = genetic_ascension_stage_1_purity
			}
		}
		bonus_pop_growth = 0.5
	}

	weight = {
		weight = @specialist_job_weight
		mult = 1.5
		mult = value:job_weights_modifier|JOB|libra_chromosome_engineer|RESOURCE|society_research|
	}

	promotion = {
		time = @standard_promotion_time
		effect = {
			pop_amount_change_category = {
				CATEGORY = specialist
				AMOUNT = local_pop_amount
			}
		}
	}
}
