libra_country_check = {
	remove_country_flag = libra_is_machine_empire
	remove_country_flag = libra_is_biological_empire
	remove_country_flag = libra_is_organic_empire
	remove_country_flag = libra_fairy_physics
	remove_country_flag = libra_fairy_engineering
	if = {
		limit = {
			or = {
				species = { is_archetype = MACHINE }
				has_authority = auth_machine_intelligence
			}
		}
		set_country_flag = libra_is_machine_empire
	}
	else = {
		set_country_flag = libra_is_biological_empire
		if = {
			limit = {
				species = {
					or = {
						is_archetype = BIOLOGICAL
						is_archetype = LITHOID
					}
				}
			}
			set_country_flag = libra_is_organic_empire
		}
	}
	random_list = {
		2 = { }
		1 = { set_country_flag = libra_fairy_physics }
		2 = { set_country_flag = libra_fairy_engineering }
	}
}

	if = {
		limit = { any_country = { is_ai = no } }
		random_country = {
			limit = { is_ai = no }
			species = { save_event_target_as = libra_agent_species }
		}
	}
	else = {
		random_country = {
			species = { save_event_target_as = libra_agent_species }
		}
	}
	create_country = {
		name = NAME_libra_agent
		type = libra_agent
		species = event_target:libra_agent_species
		origin = origin_default
		authority = auth_corporate
		ethos = {
			ethic = ethic_fanatic_xenophile
			ethic = ethic_pacifist
		}
		civics = random
		ignore_initial_colony_error = yes
		flag = {
			icon = {
				category = "spherical"
				file = "flag_spherical_6.dds"
			}
			background = {
				category = "backgrounds"
				file = "00_solid.dds"
			}
			colors={
				"green"
				"yellow"
				"null"
				"null"
			}
		}
		effect = {
			set_country_flag = libra_agent
			save_global_event_target_as = libra_agent
		}
	}
}
