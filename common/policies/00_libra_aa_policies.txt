libra_directive_76 = {
	potential = {
		has_country_flag = libra_directive_76_policy
		is_variable_set = libra_aa_path
		check_variable = { which = libra_aa_path value = 10 }
	}

	option = {
		name = policy_libra_directive_76_a
		modifier = {
			planet_soldiers_energy_produces_add = 2
			planet_soldiers_minerals_produces_add = 2
			planet_soldiers_food_produces_add = 2
		}
		AI_weight = {
			weight = 1
			modifier = {
				add = 3
				or = {
					has_deficit = energy
					has_deficit = minerals
					has_deficit = food
				}
			}
		}
	}
	option = {
		name = policy_libra_directive_76_b
		modifier = {
			planet_soldiers_unity_produces_add = 1
			planet_soldiers_trade_produces_add = 4
		}
		AI_weight = {
			weight = 0
			modifier = {
				add = 3
				libra_ai_preferences_trade = yes
			}
		}
	}
	option = {
		name = policy_libra_directive_76_c
		modifier = {
			planet_soldiers_physics_research_produces_add = 1
			planet_soldiers_society_research_produces_add = 1
			planet_soldiers_engineering_research_produces_add = 1
		}
		AI_weight = {
			weight = 2
		}
	}
}
