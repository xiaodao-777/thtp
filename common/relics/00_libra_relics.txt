@triumph_duration = 3600

r_libra_autarchy_seal = {
	activation_duration = @triumph_duration
	portrait = "GFX_relic_libra_autarchy_seal"
	sound = "event_the_great_awakening"
	can_be_stolen = no

	resources = {
		category = relics
		# Activation cost
		cost = {
			unity = @activation_cost
		}
	}

	triggered_country_modifier = {
		potential = {
			has_tradition = tr_libra_autarchy_4
			or = {
				is_galactic_emperor = yes
				galaxy_percentage > 0.95
			}
		}
		country_produces_mult = 0.15
		councilor_skill_add = 1
	}
	triggered_country_modifier = {
		potential = {
			not = {
				has_tradition = tr_libra_autarchy_4
				or = {
					is_galactic_emperor = yes
					galaxy_percentage > 0.95
				}
			}
		}
		custom_tooltip = libra_autarchy_seal_effect_tt
	}

	score = 3000

	active_effect = {
		if = {
			limit = {
				controlled_systems > 500
			}
			add_resource = { influence = 1000 trade = 250000 }
		}
		else = {
			custom_tooltip = libra_autarchy_seal_active_effect
			add_resource = {
				influence = 2
				trade = 500
				mult = trigger:controlled_systems
			}
		}
		add_modifier = {
			modifier = libra_autarchy_seal_modifier
			days = 3600
		}
		custom_tooltip = relic_triumph_cooldown
		hidden_effect = {
			add_modifier = {
				modifier = "relic_activation_cooldown"
				days = @triumph_duration
			}
		}
	}

	# Possible check for activation
	possible = {
		is_galactic_emperor = yes
		inline_script = {
			script = relics/activation_checks
			RELIC = r_libra_autarchy_seal
		}
	}

	ai_weight = {
		weight = 100
	}
}
