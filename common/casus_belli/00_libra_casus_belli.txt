cb_libra_antientropy_instruction = {
	show_in_diplomacy = no
	show_notification = no
	potential = {
		has_origin = origin_libra_antientropy_instruction
		from = { is_robot_empire = yes }
	}
	is_valid = {
		OR = {
			is_subject = no
			overlord = { NOT = { is_same_value = from } }
		}
		NOR = {
			is_overlord_to = FROM
			is_same_species = FROM
			has_valid_civic = civic_fanatic_purifiers
			has_valid_civic = civic_machine_assimilator
		}
	}
}

cb_stop_libra_antientropy_instruction = {
	show_in_diplomacy = no
	show_notification = no
	potential = {
		OR = {
			is_country_type = default
			is_country_type = awakened_fallen_empire
		}
		from = { has_origin = origin_libra_antientropy_instruction }
		not = { has_origin = origin_libra_antientropy_instruction }
		is_robot_empire = yes
	}
	is_valid = {
		has_total_war_cb = no
		OR = {
			is_subject = no
			overlord = { NOT = { is_same_value = from } }
		}
		NOR = {
			is_overlord_to = FROM
			is_same_species = FROM
		}
	}
}
