trade_action_libra_praises_to_life = {
	fire_and_forget = yes

	potential = {
		has_origin = origin_libra_praises_to_life
		from = {
			not = { has_modifier = trade_action_libra_praises_to_life_modifier }
			or = {
				is_country_type = default
				is_country_type = awakened_fallen_empire
			}
		}
	}

	on_traded_effect = {
		from = {
			add_trust = {
				amount = 36
				who = root
			}
			if = {
				limit = { has_tradition = tr_libra_ceremony_adopt }
				custom_tooltip = trade_action_libra_praises_to_life_effect
				hidden_effect = {
					change_variable = { which = libra_ceremony_count value = 2 }
				}
			}
			else = {
				add_resource = {
					influence = 50
				}
			}
			add_modifier = { modifier = trade_action_libra_praises_to_life_modifier days = 4320 }
		}
	}

	ai_weight = {
		weight = 36
	}
}
