trade_action_libra_kings_game = {
	fire_and_forget = no

	potential = {
		is_variable_set = libra_ea_path
		has_ascension_perk = ap_libra_kings_game
		has_resource = { type = unity amount > 1000 }
		check_variable = { which = libra_ea_path value = 10 }
		or = {
			is_in_federation_with = from
			has_subject = from
		}
		not = { has_modifier = trade_action_libra_kings_game_modifier }
		from = {
			not = { has_modifier = trade_action_libra_kings_game_modifier }
		}
	}

	active = {
		or = {
			is_in_federation_with = from
			has_subject = from
		}
	}

	on_traded_effect = {
		add_resource = {
			unity = -500
		}
		add_modifier = { modifier = trade_action_libra_kings_game_modifier }
		from = {
			add_modifier = { modifier = trade_action_libra_kings_game_modifier }
		}
	}

	on_deal_ended_sender_effect = {
		if = {
			limit = {
				not = {
					any_country = {
						is_action_active = {
							action = trade_action_libra_kings_game
							with_country = root
						}
					}
				}
			}
			remove_modifier = trade_action_libra_kings_game_modifier
		}
	}

	on_deal_ended_recipient_effect = {
		remove_modifier = trade_action_libra_kings_game_modifier
	}

	ai_weight = {
		weight = 2
	}
}
